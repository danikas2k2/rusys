import RemoveIcon from '@icons/Remove.svg';
import Button from '@ui/Button';
import Menu, { MenuButton } from '@ui/Menu';
import React from 'react';

export const colors = ['neutral', 'primary', 'secondary', 'positive', 'warning', 'negative'];
export const variants = ['solid', 'soft', 'outlined', 'plain'];
export const sizes = ['small', 'medium', 'large'];

# Menu

```tsx
import Menu from '@ui/Menu';

<Menu
    variant="solid"
    color="neutral"
    size="small"
    spacing="small"
    disabled={false}
    // ...
    // any other valid html attributes, applied to the HTMLButtonElement
>
    Content
</Menu>;
```

| Property   | Type                                                                                     | Default     | Description                     |
| :--------- | :--------------------------------------------------------------------------------------- | :---------- | :------------------------------ |
| `variant`  | `"solid"` \| `"soft"` \| `"outlined"` \| `"plain"`                                       | `"solid"`   | The variant of the button.      |
| `color`    | `"primary"` \| `"secondary"` \| `"neutral"` \| `"positive"` \| `"warning"` \| `"negative"` | `"neutral"` | The color of the button.        |
| `size`     | `"small"` \| `"medium"` \| `"large"`                                                     | `"small"`   | The size of the button.         |
| `spacing`  | `"small"` \| `"medium"` \| `"large"` \| `"none"`                                         | `"small"`   | The spacing of the button.      |
| `disabled` | `boolean`                                                                                | `false`     | Whether the button is disabled. |
