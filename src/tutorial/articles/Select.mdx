export const colors = ['neutral', 'primary', 'secondary', 'positive', 'warning', 'negative'];
export const variants = ['outlined', 'plain', 'soft', 'solid'];
export const sizes = ['small', 'medium', 'large'];
export const spacing = ['small', 'medium', 'large', 'half', 'none'];
export const modes = ['single', 'multiple'];

# Select

```tsx
import { Select } from '@ui/Select';

<Select
    variant="outlined"
    color="neutral"
    size="small"
    spacing="small"
    mode="single"
    value=""
    placeholder="Select something"
    disabled={false}
    fullWidth={false}
    // ...
    // any other valid html attributes, applied to the HTMLInputElement
/>;
```

| Property      | Type                                                                               | Default        | Description                                                          |
| :------------ | :--------------------------------------------------------------------------------- | :------------- | :------------------------------------------------------------------- |
| `variant`     | _`"solid"`, `"soft"`, `"outlined"`, `"plain"`_                                     | _`"outlined"`_ | The variant of the select.                                           |
| `color`       | _`"primary"`, `"secondary"`, `"neutral"`, `"positive"`, `"warning"`, `"negative"`_ | _`"neutral"`_  | The color of the select.                                             |
| `size`        | _`"small"`, `"medium"`, `"large"`_                                                 | _`"small"`_    | The size of the select.                                              |
| `spacing`     | _`"small"`, `"medium"`, `"large"`_                                                 | _`"small"`_    | The spacing of the select.                                           |
| `mode`        | _`"single"`, `"multiple"`_                                                         | _`"single"`_   | The mode of the select.                                              |
| `value`       | `string`, `number`                                                                 | -              | The value of the select.                                             |
| `placeholder` | `string`, `number`                                                                 | -              | The placeholder of the select.                                       |
| `disabled`    | `boolean`                                                                          | `false`        | Whether the input is disabled.                                       |
| `fullWidth`   | `boolean`                                                                          | `false`        | Whether the input is full width.                                     |
| `label`       | `string`                                                                           | -              | The label input.                                                     |
| `placeholder` | `string`                                                                           | -              | The placeholder of the select, label is used if placeholder omitted. |
| `error`       | `boolean`                                                                          | `false`        | Whether the input has error.                                         |
| `errorLabel`  | `string`                                                                           | -              | The friendly error label.                                            |

export const options = [
