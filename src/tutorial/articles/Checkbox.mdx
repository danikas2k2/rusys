import { Checkbox } from '@ui/Checkbox';
import React from 'react';

export const colors = ['neutral', 'primary', 'secondary', 'positive', 'warning', 'negative'];
export const variants = ['outlined', 'plain', 'soft', 'solid'];
export const sizes = ['small', 'medium', 'large'];

# Checkbox

```tsx
import { Checkbox } from '@ui/Checkbox';

<Checkbox
    variant="outlined"
    color="neutral"
    size="small"
    checked={false}
    indeterminate={false}
    disabled={false}
    // ...
    // any other valid html attributes, applied to the HTMLInputElement
>
    Label
</Checkbox>;
```

| Property        | Type                                                                                       | Default      | Description                            |
| :-------------- | :----------------------------------------------------------------------------------------- | :----------- | :------------------------------------- |
| `variant`       | `"solid"` \| `"soft"` \| `"outlined"` \| `"plain"`                                         | `"outlined"` | The variant of the checkbox.           |
| `color`         | `"primary"` \| `"secondary"` \| `"neutral"` \| `"positive"` \| `"warning"` \| `"negative"` | `"neutral"`  | The color of the checkbox.             |
| `size`          | `"small"` \| `"medium"` \| `"large"`                                                       | `"small"`    | The size of the checkbox.              |
| `spacing`       | `"small"` \| `"medium"` \| `"large"` \| `"none"`                                           | `"small"`    | The spacing of the checkbox.           |
| `checked`       | `boolean`                                                                                  | `false`      | Whether the checkbox is checked.       |
| `indeterminate` | `boolean`                                                                                  | `false`      | Whether the checkbox is indeterminate. |
| `disabled`      | `boolean`                                                                                  | `false`      | Whether the checkbox is disabled.      |
