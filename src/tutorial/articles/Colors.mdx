import React from 'react';
import { generate } from 'random-words';

export const colors = ['neutral', 'primary', 'secondary', 'positive', 'warning', 'negative'];
export const variants = ['solid', 'soft', 'outlined', 'plain'];
export const steps = ['50', '100', '200', '300', '400', '500', '600', '700', '800', '900'];
export const states = ['', 'hover', 'active', 'disabled'];

export const name = (...parts) => `--${parts.filter(Boolean).join('-')}`;
export const variable = (...parts) => {
    const filtered = parts.filter(Boolean);
    const prop = ['bg', 'border'].includes(filtered[filtered.length - 1]);
    const offset = filtered.length - prop - 1;
    const state = states.includes(filtered[offset]);
    return `var(${name(...filtered)}${
        state ? `, ${variable(...filtered.slice(0, offset), ...filtered.slice(offset + 1))}` : ''
    })`;
};
export const paletteVariable = (...parts) => variable('palette', ...parts);
export const colorVariable = (...parts) => variable('color', ...parts);

export const short = () => generate({ exactly: 1, minLength: 5, maxLength: 10, join: ' ' });
export const long = () => generate({ min: 3, max: 10, maxLength: 50, join: ' ' });

export function Box({ color, background, border, children = short() }) {
    return (
        <div
            style={{
                color,
                background,
                border: `1px solid ${border}`,
                borderRadius: 4,
                padding: '1rem .25rem',
                textAlign: 'center',
                whiteSpace: 'nowrap',
                overflow: 'hidden',
                textOverflow: 'ellipsis',
            }}
        >
            {children}
        </div>
    );
}

# Colors

## Palette variables

Palette variables consists of starting `--palette` prefix followed by **group** part and
**darkness** weight, separated by dash. There are two special group, names **common**.
There are two palette values in this group: **white** and **black**.

```css
:root {
    --color-text-primary: var(--palette-neutral-800);
    --color-background-body: var(--palette-common-white);
}
```

## Predefined variables

Predefined color variables, like palette variables, consists of starting `--color` prefix followed by **color** and
**variant** parts, then optional **state** part may be applied. Finally, **bg** or **border** part may be added.
All parts are dash separated. It's recommended to use default values for state variables, if not defined in upper scope.

```css
input {
    color: var(--color-neutral-solid);
    background-color: var(--color-primary-soft-bg);
    border-color: var(--color-positive-outlined-border);

    &:disabled {
        color: var(--color-neutral-soft-disabled, var(--color-neutral-soft));
        background-color: var(--color-primary-solid-disabled-bg, var(--color-primary-solid-bg));
        border-color: var(--color-positive-outlined-disabled-border, var(--color-positive-outlined-border));
    }

    &:hover {
        color: var(--color-neutral-soft-hover, var(--color-neutral-soft));
        background-color: var(--color-primary-solid-hover-bg, var(--color-primary-solid-bg));
        border-color: var(--color-positive-outlined-hover-border, var(--color-positive-outlined-border));
    }

    &:active {
        color: var(--color-neutral-soft-active, var(--color-neutral-soft));
        background-color: var(--color-primary-solid-active-bg, var(--color-primary-solid-bg));
    }
}
```
